cmake_minimum_required(VERSION 3.20)
project(OverkillFoundryStorage LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Only the verified Windows storage backend is implemented.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(overkill_save_store STATIC src/windows_save_store.cpp)
target_include_directories(overkill_save_store PUBLIC include)
target_link_libraries(overkill_save_store PRIVATE bcrypt)
target_compile_definitions(overkill_save_store PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(overkill_save_store PRIVATE /W4 /WX /permissive-)
add_executable(overkill_storage_tests tests/storage_tests.cpp)
target_link_libraries(overkill_storage_tests PRIVATE overkill_save_store)
enable_testing()
add_test(NAME storage_contracts COMMAND overkill_storage_tests)
set(OVERKILL_BUILD_CORE_TOOLS OFF CACHE BOOL "Storage build uses only the shared rules library" FORCE)
add_subdirectory(../core core-runtime EXCLUDE_FROM_ALL)
add_library(overkill_campaign_session STATIC src/campaign_session.cpp)
target_include_directories(overkill_campaign_session PUBLIC include)
target_link_libraries(overkill_campaign_session PUBLIC overkill_core overkill_save_store)
target_compile_options(overkill_campaign_session PRIVATE /W4 /WX /permissive-)
add_executable(overkill_campaign_storage_tests tests/campaign_storage_tests.cpp)
target_link_libraries(overkill_campaign_storage_tests PRIVATE overkill_campaign_session)
add_test(NAME campaign_storage_contracts COMMAND overkill_campaign_storage_tests)
