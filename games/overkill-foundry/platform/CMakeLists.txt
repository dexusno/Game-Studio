cmake_minimum_required(VERSION 3.20)
project(OverkillFoundryStorage LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Only the verified Windows storage backend is implemented.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(overkill_save_store STATIC src/windows_save_store.cpp)
target_include_directories(overkill_save_store PUBLIC include)
target_link_libraries(overkill_save_store PRIVATE bcrypt)
target_compile_definitions(overkill_save_store PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(overkill_save_store PRIVATE /W4 /WX /permissive-)
add_executable(overkill_storage_tests tests/storage_tests.cpp)
target_link_libraries(overkill_storage_tests PRIVATE overkill_save_store)
enable_testing()
add_test(NAME storage_contracts COMMAND overkill_storage_tests)
