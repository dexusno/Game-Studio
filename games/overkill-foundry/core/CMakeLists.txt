cmake_minimum_required(VERSION 3.20)
project(OverkillFoundryCore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(overkill_core STATIC src/core.cpp src/serialization.cpp)
target_include_directories(overkill_core PUBLIC include)
if(MSVC)
  target_compile_options(overkill_core PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(overkill_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(overkill_runner runner/main.cpp)
target_link_libraries(overkill_runner PRIVATE overkill_core)
add_executable(overkill_core_tests tests/core_tests.cpp)
target_link_libraries(overkill_core_tests PRIVATE overkill_core)
enable_testing()
add_test(NAME core_contracts COMMAND overkill_core_tests)
