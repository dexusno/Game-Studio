cmake_minimum_required(VERSION 3.20)
project(OverkillFoundryCore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(overkill_core STATIC src/core.cpp src/serialization.cpp src/catalogue.cpp src/robots.cpp src/route.cpp src/city_content.cpp src/campaign.cpp src/campaign_serialization.cpp src/campaign_upgrades.cpp src/upgrades.cpp)
target_include_directories(overkill_core PUBLIC include)
if(MSVC)
  target_compile_options(overkill_core PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(overkill_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(OVERKILL_BUILD_CORE_TOOLS "Build standalone runner and core contract tests" ON)
if(OVERKILL_BUILD_CORE_TOOLS)
add_executable(overkill_runner runner/main.cpp)
target_link_libraries(overkill_runner PRIVATE overkill_core)
add_executable(overkill_core_tests tests/core_tests.cpp)
target_link_libraries(overkill_core_tests PRIVATE overkill_core)
enable_testing()
add_test(NAME core_contracts COMMAND overkill_core_tests)

add_executable(overkill_route_tests tests/route_tests.cpp)
target_link_libraries(overkill_route_tests PRIVATE overkill_core)
add_test(NAME route_contracts COMMAND overkill_route_tests)
add_executable(overkill_robot_tests tests/robot_tests.cpp)
target_link_libraries(overkill_robot_tests PRIVATE overkill_core)
target_compile_definitions(overkill_robot_tests PRIVATE OVERKILL_ROBOT_INTEGRATION=1)
add_test(NAME robot_contracts COMMAND overkill_robot_tests)

add_executable(overkill_recipe_tests tests/recipe_tests.cpp)
target_link_libraries(overkill_recipe_tests PRIVATE overkill_core)
add_test(NAME recipe_contracts COMMAND overkill_recipe_tests)

add_executable(overkill_upgrade_tests tests/upgrade_tests.cpp)
target_link_libraries(overkill_upgrade_tests PRIVATE overkill_core)
add_test(NAME upgrade_contracts COMMAND overkill_upgrade_tests)

add_executable(overkill_campaign_tests tests/campaign_tests.cpp)
target_link_libraries(overkill_campaign_tests PRIVATE overkill_core)
add_test(NAME campaign_contracts COMMAND overkill_campaign_tests)
add_executable(overkill_campaign_upgrade_tests tests/campaign_upgrade_tests.cpp)
target_link_libraries(overkill_campaign_upgrade_tests PRIVATE overkill_core)
add_test(NAME campaign_upgrade_contracts COMMAND overkill_campaign_upgrade_tests)
endif()
