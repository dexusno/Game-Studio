cmake_minimum_required(VERSION 3.20)
project(PartLifecycleContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OVERKILL_BUILD_CORE_TOOLS OFF CACHE BOOL "Isolated lifecycle evidence" FORCE)
add_subdirectory(../.. core-runtime EXCLUDE_FROM_ALL)
add_executable(part_lifecycle_tests ../part_lifecycle_tests.cpp)
target_link_libraries(part_lifecycle_tests PRIVATE overkill_core)
if(MSVC)
  target_compile_options(part_lifecycle_tests PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(part_lifecycle_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME physical_part_lifecycle COMMAND part_lifecycle_tests)
