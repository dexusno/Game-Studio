cmake_minimum_required(VERSION 3.20)
project(SharedRecipeSemanticContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OVERKILL_BUILD_CORE_TOOLS OFF CACHE BOOL "Isolated Shared recipe evidence" FORCE)
add_subdirectory(../.. core-runtime EXCLUDE_FROM_ALL)
add_executable(shared_recipe_contract_tests ../shared_recipe_contract_tests.cpp)
add_executable(shared_recipe_legacy_tests ../recipe_tests.cpp)
foreach(target shared_recipe_contract_tests shared_recipe_legacy_tests)
  target_link_libraries(${target} PRIVATE overkill_core)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME shared_recipe_contracts COMMAND shared_recipe_contract_tests)
add_test(NAME existing_recipe_regression COMMAND shared_recipe_legacy_tests)
