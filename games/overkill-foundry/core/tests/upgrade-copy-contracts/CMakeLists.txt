cmake_minimum_required(VERSION 3.20)
project(UpgradeCopyContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OVERKILL_BUILD_CORE_TOOLS OFF CACHE BOOL "Isolated single-output attribution regression" FORCE)
add_subdirectory(../.. core-runtime EXCLUDE_FROM_ALL)
add_executable(upgrade_copy_contract_tests ../upgrade_copy_contract_tests.cpp)
add_executable(upgrade_copy_legacy_tests ../upgrade_tests.cpp)
add_executable(upgrade_copy_author_tests ../upgrade_contract_tests.cpp)
add_executable(upgrade_copy_shared_tests ../shared_recipe_contract_tests.cpp)
add_executable(upgrade_copy_part_tests ../part_lifecycle_tests.cpp)
foreach(target upgrade_copy_contract_tests upgrade_copy_legacy_tests upgrade_copy_author_tests upgrade_copy_shared_tests upgrade_copy_part_tests)
  target_link_libraries(${target} PRIVATE overkill_core)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME output_copy_attribution COMMAND upgrade_copy_contract_tests)
add_test(NAME existing_upgrade_regression COMMAND upgrade_copy_legacy_tests)
add_test(NAME existing_author_upgrade_regression COMMAND upgrade_copy_author_tests)
add_test(NAME existing_shared_recipe_regression COMMAND upgrade_copy_shared_tests)
add_test(NAME existing_physical_lifecycle_regression COMMAND upgrade_copy_part_tests)
