cmake_minimum_required(VERSION 3.20)
project(TerminalEventContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OVERKILL_BUILD_CORE_TOOLS OFF CACHE BOOL "Isolated terminal regression" FORCE)
add_subdirectory(../.. core-runtime EXCLUDE_FROM_ALL)
add_executable(terminal_tests terminal_tests.cpp)
target_link_libraries(terminal_tests PRIVATE overkill_core)
if(MSVC)
  target_compile_options(terminal_tests PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(terminal_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME terminal_event_transitions COMMAND terminal_tests)
