cmake_minimum_required(VERSION 3.20)
project(UpgradeSemanticContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OVERKILL_BUILD_CORE_TOOLS OFF CACHE BOOL "Isolated upgrade semantic evidence" FORCE)
add_subdirectory(../.. core-runtime EXCLUDE_FROM_ALL)
add_executable(upgrade_contract_tests ../upgrade_contract_tests.cpp)
add_executable(upgrade_contract_legacy_upgrades ../upgrade_tests.cpp)
add_executable(upgrade_contract_campaign_upgrades ../campaign_upgrade_tests.cpp)
add_executable(upgrade_contract_campaign ../campaign_tests.cpp)
foreach(target upgrade_contract_tests upgrade_contract_legacy_upgrades upgrade_contract_campaign_upgrades upgrade_contract_campaign)
  target_link_libraries(${target} PRIVATE overkill_core)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME upgrade_semantic_contracts COMMAND upgrade_contract_tests)
add_test(NAME existing_upgrade_regression COMMAND upgrade_contract_legacy_upgrades)
add_test(NAME existing_upgrade_campaign_regression COMMAND upgrade_contract_campaign_upgrades)
add_test(NAME existing_campaign_regression COMMAND upgrade_contract_campaign)
