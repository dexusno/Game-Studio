cmake_minimum_required(VERSION 3.20)
project(OverkillUpgradeContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(upgrade_core STATIC ../../src/core.cpp ../../src/serialization.cpp ../../src/catalogue.cpp ../../src/robots.cpp ../../src/upgrades.cpp)
target_include_directories(upgrade_core PUBLIC ../../include)
if(MSVC)
  target_compile_options(upgrade_core PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(upgrade_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
foreach(name core recipe robot upgrade)
  add_executable(overkill_${name}_tests ../${name}_tests.cpp)
  target_link_libraries(overkill_${name}_tests PRIVATE upgrade_core)
  add_test(NAME ${name}_contracts COMMAND overkill_${name}_tests)
endforeach()
target_compile_definitions(overkill_robot_tests PRIVATE OVERKILL_ROBOT_INTEGRATION=1)
