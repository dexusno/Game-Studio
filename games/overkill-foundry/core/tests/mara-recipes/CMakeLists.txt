cmake_minimum_required(VERSION 3.20)
project(MaraRecipeSemanticContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(MARA_CORE_ROOT "" CACHE PATH "Explicit captured authoritative core")
if(NOT MARA_CORE_ROOT)
  get_filename_component(MARA_CORE_ROOT "../.." ABSOLUTE)
endif()
set(core_sources core serialization catalogue robots upgrades route city_content campaign campaign_upgrades campaign_serialization)
list(TRANSFORM core_sources PREPEND "${MARA_CORE_ROOT}/src/")
list(TRANSFORM core_sources APPEND ".cpp")
add_library(mara_contract_core STATIC ${core_sources})
target_include_directories(mara_contract_core PUBLIC "${MARA_CORE_ROOT}/include")
add_executable(mara_recipe_contract_tests ../mara_recipe_contract_tests.cpp)
target_link_libraries(mara_recipe_contract_tests PRIVATE mara_contract_core)
add_executable(mara_legacy_recipe_tests ../recipe_tests.cpp)
target_link_libraries(mara_legacy_recipe_tests PRIVATE mara_contract_core)
foreach(target mara_contract_core mara_recipe_contract_tests mara_legacy_recipe_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME mara_recipe_semantic_contracts COMMAND mara_recipe_contract_tests)
add_test(NAME existing_recipe_regression COMMAND mara_legacy_recipe_tests)
