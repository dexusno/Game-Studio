cmake_minimum_required(VERSION 3.20)
project(OverkillFoundryRobotContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(overkill_robot_module_tests ../robot_tests.cpp ../../src/robots.cpp)
target_include_directories(overkill_robot_module_tests PRIVATE ../../include)
if(MSVC)
  target_compile_options(overkill_robot_module_tests PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(overkill_robot_module_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME robot_module_contracts COMMAND overkill_robot_module_tests)
