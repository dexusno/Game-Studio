cmake_minimum_required(VERSION 3.20)
project(FoundryEncounterContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OVERKILL_BUILD_CORE_TOOLS OFF CACHE BOOL "Isolated encounter contracts" FORCE)
add_subdirectory(../.. core-runtime EXCLUDE_FROM_ALL)
add_executable(encounter_contract_tests ../encounter_contract_tests.cpp)
target_link_libraries(encounter_contract_tests PRIVATE overkill_core)
if(MSVC)
  target_compile_options(encounter_contract_tests PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(encounter_contract_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME encounter_semantics COMMAND encounter_contract_tests)
