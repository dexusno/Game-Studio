cmake_minimum_required(VERSION 3.20)
project(IndependentEncounterLifecycleReview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED FROZEN_GAME)
  message(FATAL_ERROR "Pass one exact archived game source directory")
endif()
option(QA_CAMPAIGN "Review campaign-lifecycle graph instead of encounter graph" OFF)
set(OVERKILL_BUILD_CORE_TOOLS OFF CACHE BOOL "Independent frozen graph" FORCE)
add_subdirectory("${FROZEN_GAME}/core" core-runtime EXCLUDE_FROM_ALL)
if(QA_CAMPAIGN)
  add_executable(independent_probes campaign_probes.cpp)
else()
  add_executable(independent_probes encounter_probes.cpp)
endif()
target_link_libraries(independent_probes PRIVATE overkill_core)
if(MSVC)
  target_compile_options(independent_probes PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(independent_probes PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME independent_boundaries COMMAND independent_probes)
