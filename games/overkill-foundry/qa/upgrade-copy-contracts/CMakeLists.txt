cmake_minimum_required(VERSION 3.20)
project(IndependentUpgradeCopies LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT FROZEN_GAME OR NOT FROZEN_LIBRARY OR NOT ORIGINAL_QA_SOURCE)
  message(FATAL_ERROR "Provide exact frozen headers, rebuilt library and unchanged original QA source")
endif()
add_library(frozen_core STATIC IMPORTED)
set_target_properties(frozen_core PROPERTIES IMPORTED_LOCATION "${FROZEN_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${FROZEN_GAME}/core/include")
add_executable(independent_upgrade_copies probes.cpp)
add_executable(original_upgrade_review "${ORIGINAL_QA_SOURCE}")
foreach(target independent_upgrade_copies original_upgrade_review)
  target_link_libraries(${target} PRIVATE frozen_core)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
endforeach()
if(LEGACY_GAME AND LEGACY_LIBRARY)
  add_library(legacy_core STATIC IMPORTED)
  set_target_properties(legacy_core PROPERTIES IMPORTED_LOCATION "${LEGACY_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${LEGACY_GAME}/core/include")
  add_executable(legacy_state_emitter legacy-emitter.cpp)
  target_link_libraries(legacy_state_emitter PRIVATE legacy_core)
  target_compile_options(legacy_state_emitter PRIVATE /W4 /WX /permissive- /utf-8)
endif()
