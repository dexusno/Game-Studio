cmake_minimum_required(VERSION 3.20)
project(IndependentUpgradeContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT FROZEN_GAME OR NOT FROZEN_LIBRARY)
  message(FATAL_ERROR "Provide the captured input game directory and independently rebuilt core library")
endif()
add_library(frozen_core STATIC IMPORTED)
set_target_properties(frozen_core PROPERTIES IMPORTED_LOCATION "${FROZEN_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${FROZEN_GAME}/core/include")
add_executable(independent_upgrade_contracts probes.cpp)
target_link_libraries(independent_upgrade_contracts PRIVATE frozen_core)
target_compile_options(independent_upgrade_contracts PRIVATE /W4 /WX /permissive- /utf-8)
