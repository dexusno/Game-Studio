cmake_minimum_required(VERSION 3.20)
project(IndependentPartReview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT DEFINED FROZEN_INPUTS)
  message(FATAL_ERROR "Supply the captured lifecycle inputs, never the changing live core")
endif()
set(OVERKILL_BUILD_CORE_TOOLS OFF CACHE BOOL "Isolated independent review" FORCE)
add_subdirectory("${FROZEN_INPUTS}/core" frozen-core EXCLUDE_FROM_ALL)
add_executable(independent_part_probes probes.cpp)
target_link_libraries(independent_part_probes PRIVATE overkill_core)
add_executable(rebuilt_author_parts "${FROZEN_INPUTS}/core/tests/part_lifecycle_tests.cpp")
target_link_libraries(rebuilt_author_parts PRIVATE overkill_core)
foreach(target independent_part_probes rebuilt_author_parts)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
enable_testing()
add_test(NAME independent_parts COMMAND independent_part_probes)
add_test(NAME captured_author_parts COMMAND rebuilt_author_parts)
