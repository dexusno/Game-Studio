cmake_minimum_required(VERSION 3.20)
project(IndependentSharedRecipes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT FROZEN_GAME OR NOT FROZEN_LIBRARY)
  message(FATAL_ERROR "Provide exact captured inputs and rebuilt frozen library")
endif()
add_library(frozen_core STATIC IMPORTED)
set_target_properties(frozen_core PROPERTIES IMPORTED_LOCATION "${FROZEN_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${FROZEN_GAME}/core/include")
add_executable(independent_shared_recipes probes.cpp)
target_link_libraries(independent_shared_recipes PRIVATE frozen_core)
target_compile_options(independent_shared_recipes PRIVATE /W4 /WX /permissive- /utf-8)
