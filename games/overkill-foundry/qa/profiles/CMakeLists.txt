cmake_minimum_required(VERSION 3.20)
project(IndependentProfiles LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT FROZEN_GAME)
  message(FATAL_ERROR "Set FROZEN_GAME to the preserved 2016 game sources")
endif()
set(core_files core serialization catalogue robots route city_content campaign campaign_serialization campaign_upgrades upgrades)
list(TRANSFORM core_files PREPEND "${FROZEN_GAME}/core/src/")
list(TRANSFORM core_files APPEND ".cpp")
add_library(profile_core STATIC ${core_files})
target_include_directories(profile_core PUBLIC "${FROZEN_GAME}/core/include")
add_library(profile_storage STATIC "${FROZEN_GAME}/platform/src/windows_save_store.cpp" "${FROZEN_GAME}/platform/src/campaign_session.cpp")
target_include_directories(profile_storage PUBLIC "${FROZEN_GAME}/platform/include")
target_compile_definitions(profile_storage PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(profile_storage PUBLIC profile_core PRIVATE bcrypt)
if(NOT PROFILE_SOURCE_OVERRIDE)
  set(PROFILE_SOURCE_OVERRIDE "${FROZEN_GAME}/unreal/Source/OverkillFoundry/Private/FoundryProfiles.cpp")
endif()
add_executable(independent_profiles probes.cpp "${PROFILE_SOURCE_OVERRIDE}")
target_include_directories(independent_profiles PRIVATE "${FROZEN_GAME}/unreal/Source/OverkillFoundry/Public")
target_compile_definitions(independent_profiles PRIVATE FOUNDRY_WITH_CAMPAIGN=1 NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(independent_profiles PRIVATE profile_storage)
foreach(target profile_core profile_storage independent_profiles)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
endforeach()
