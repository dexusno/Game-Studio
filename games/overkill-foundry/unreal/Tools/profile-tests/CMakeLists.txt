cmake_minimum_required(VERSION 3.20)
project(FoundryProfiles LANGUAGES CXX)
set(FOUNDRY_TEST_CXX_STANDARD 17 CACHE STRING "C++ standard (17 standalone, 20 Unreal compatibility)")
set(CMAKE_CXX_STANDARD ${FOUNDRY_TEST_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(FOUNDRY_TEST_CORE_ROOT "" CACHE PATH "Explicit reviewed core snapshot; empty uses the current core")
if(NOT FOUNDRY_TEST_CORE_ROOT)
  get_filename_component(FOUNDRY_TEST_CORE_ROOT "../../../core" ABSOLUTE)
endif()
set(core_sources core serialization catalogue robots route city_content campaign campaign_serialization campaign_upgrades upgrades)
list(TRANSFORM core_sources PREPEND "${FOUNDRY_TEST_CORE_ROOT}/src/")
list(TRANSFORM core_sources APPEND ".cpp")
add_library(profile_core STATIC ${core_sources})
target_include_directories(profile_core PUBLIC "${FOUNDRY_TEST_CORE_ROOT}/include")
add_library(profile_storage STATIC ../../../platform/src/windows_save_store.cpp ../../../platform/src/campaign_session.cpp)
target_include_directories(profile_storage PUBLIC ../../../platform/include)
target_compile_definitions(profile_storage PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(profile_storage PUBLIC profile_core PRIVATE bcrypt)
add_executable(foundry_profiles main.cpp ../../Source/OverkillFoundry/Private/FoundryProfiles.cpp)
target_include_directories(foundry_profiles PRIVATE ../../Source/OverkillFoundry/Public)
target_compile_definitions(foundry_profiles PRIVATE FOUNDRY_WITH_CAMPAIGN=1)
target_link_libraries(foundry_profiles PRIVATE profile_storage)
add_executable(foundry_profile_fixtures fixtures.cpp)
target_link_libraries(foundry_profile_fixtures PRIVATE profile_storage)
foreach(target profile_core profile_storage foundry_profiles foundry_profile_fixtures)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
endforeach()
enable_testing()
add_test(NAME profile_contracts COMMAND foundry_profiles)
