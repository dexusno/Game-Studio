cmake_minimum_required(VERSION 3.20)
project(FoundryCityReplay LANGUAGES CXX)
set(FOUNDRY_CITY_CXX_STANDARD 17 CACHE STRING "17 standalone, 20 Unreal compatibility")
set(CMAKE_CXX_STANDARD ${FOUNDRY_CITY_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(FOUNDRY_CITY_CORE_ROOT "" CACHE PATH "Required explicit reviewed core snapshot")
if(NOT FOUNDRY_CITY_CORE_ROOT)
  message(FATAL_ERROR "Pass the explicit reviewed 1843 core snapshot as FOUNDRY_CITY_CORE_ROOT")
endif()
get_filename_component(game_root "../../.." ABSOLUTE)
set(core_sources core serialization catalogue robots route city_content campaign campaign_serialization campaign_upgrades upgrades)
list(TRANSFORM core_sources PREPEND "${FOUNDRY_CITY_CORE_ROOT}/src/")
list(TRANSFORM core_sources APPEND ".cpp")
add_library(city_core STATIC ${core_sources})
target_include_directories(city_core PUBLIC "${FOUNDRY_CITY_CORE_ROOT}/include")
add_library(city_storage STATIC "${game_root}/platform/src/windows_save_store.cpp" "${game_root}/platform/src/campaign_session.cpp")
target_include_directories(city_storage PUBLIC "${game_root}/platform/include")
target_compile_definitions(city_storage PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(city_storage PUBLIC city_core PRIVATE bcrypt)
add_library(city_replay STATIC replay.cpp)
target_link_libraries(city_replay PUBLIC city_core)
# Link the original historical codec for an independent field-order check.
# Its policy is a link dependency of trace.cpp; the preparation executable never
# calls Policy::choose/runCity. The Unreal probe does not link either source.
add_executable(foundry_city_prepare main.cpp "${game_root}/core/runner/trace.cpp" "${game_root}/core/runner/policy.cpp")
target_include_directories(foundry_city_prepare PRIVATE "${game_root}/core/runner")
target_link_libraries(foundry_city_prepare PRIVATE city_replay city_storage)
foreach(target city_core city_storage city_replay foundry_city_prepare)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
endforeach()
