cmake_minimum_required(VERSION 3.20)
project(FoundryRecipeControls LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OVERKILL_BUILD_CORE_TOOLS OFF CACHE BOOL "Focused presentation test" FORCE)
set(FOUNDRY_TEST_CORE_ROOT "" CACHE PATH "Explicit reviewed core source snapshot; empty uses the current shared core")
if(FOUNDRY_TEST_CORE_ROOT)
  set(core_sources core serialization catalogue robots route city_content campaign campaign_serialization campaign_upgrades upgrades)
  list(TRANSFORM core_sources PREPEND "${FOUNDRY_TEST_CORE_ROOT}/src/")
  list(TRANSFORM core_sources APPEND ".cpp")
  add_library(overkill_core STATIC ${core_sources})
  target_include_directories(overkill_core PUBLIC "${FOUNDRY_TEST_CORE_ROOT}/include")
  if(MSVC)
    target_compile_options(overkill_core PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(overkill_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
else()
  add_subdirectory(../../../core core EXCLUDE_FROM_ALL)
endif()
add_executable(foundry_recipe_controls main.cpp ../../Source/OverkillFoundry/Private/FoundryRecipeControls.cpp)
target_include_directories(foundry_recipe_controls PRIVATE ../../Source/OverkillFoundry/Public)
target_compile_definitions(foundry_recipe_controls PRIVATE FOUNDRY_WITH_CAMPAIGN=1)
target_link_libraries(foundry_recipe_controls PRIVATE overkill_core)
if(MSVC)
  target_compile_options(foundry_recipe_controls PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(foundry_recipe_controls PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME recipe_control_contracts COMMAND foundry_recipe_controls)
