cmake_minimum_required(VERSION 3.20)
project(FoundryPrecisionTests LANGUAGES CXX)
enable_testing()
add_executable(foundry_precision_tests tests.cpp)
target_compile_features(foundry_precision_tests PRIVATE cxx_std_17)
target_include_directories(foundry_precision_tests PRIVATE ../../../presentation)
if(MSVC)
  target_compile_options(foundry_precision_tests PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(foundry_precision_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME precision_input_contract COMMAND foundry_precision_tests)
